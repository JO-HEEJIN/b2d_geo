## 1. 컨설팅 반영: 올맵 리포트 화면 (10-24, ad8f808 외)

### 요구 (황대표님 화이트보드)

1. 서비스명 "올맵"
2. 화면 4열: [항공사진 + 지도 1/3 | 거래내역(주변 지역, 시간순) | 지가변동률(X·Y축, 범위 명시) | 판단 요약 + 출력 박스]
3. 단위: ㎡(헤베)와 평(3.3㎡ = 1평) 병기
4. 보고서 자동 작성
5. 월요일(10-26) 다나클라우드 미팅 전 배포

### 결정과 그 이유

- 거리뷰는 넣지 않았다. 지시서 1.1이 카카오·네이버 로드뷰 사용을 금지한다(상용 판매 시 라이선스 문제). 항공사진은
  지시서 허용 출처인 V-World WMTS 영상 타일로 대신했다.
- 실거래는 필지 핀이 아니라 법정동 대표점 마커다. 국토부 토지 실거래는 지번이 마스킹되어 온다.
- 실거래를 브라우저가 월별로 여러 번 부르면 demo 키(2 rps, burst 4)에서 429가 난다. 그래서 서버에
  `/v1/appraisal/sales/recent`를 만들어 최근 N개월(기본 3, 최대 6)을 동시에 모아 한 번에 준다.
- 보고서는 서버나 LLM 없이 브라우저 인쇄(A4 print-mode)로 만든다. "PDF로 저장"을 고르면 파일이 된다.

법정동 대표점. `admin_areas.geom`이 운영 DB에서 전부 NULL이라(1차 QA에서 발견) 표준지 좌표의 법정동별 평균으로 구한다:

```go
// 범위 조건으로 인덱스를 탄다(hasRegion과 같은 이유). 시군구 자체 행(xxxxx00000)은 표준지가
// 없어 자연히 빠진다. 이름은 admin_areas 원문("시도 시군구 읍면동")이고, 프론트가 MOLIT
// umdNm과 접미 일치로 맞춘다.
func (s *Server) umdPoints(ctx context.Context, lawd string) ([]umdPoint, error) {
	if s.Pool == nil {
		return []umdPoint{}, nil
	}
	rows, err := s.Pool.Query(ctx,
		`SELECT a.name, s.lng, s.lat
		   FROM (SELECT substr(pnu, 1, 10) AS bcode,
		                avg(ST_X(geom)) AS lng, avg(ST_Y(geom)) AS lat
		           FROM (SELECT DISTINCT ON (pnu) pnu, geom
		                   FROM standard_lots
		                  WHERE pnu >= $1 AND pnu <= $2
		                  ORDER BY pnu, base_year DESC) latest
		          GROUP BY 1) s
		   JOIN admin_areas a ON a.bcode = s.bcode
		  ORDER BY s.bcode
		  LIMIT $3`,
		lawd+"00000000000000", lawd+"99999999999999", umdPointsLimit)
	if err != nil {
		return nil, err
	}
	defer rows.Close()
	out := []umdPoint{}
	for rows.Next() {
		var p umdPoint
		if err := rows.Scan(&p.Name, &p.Lng, &p.Lat); err != nil {
			return nil, err
		}
		out = append(out, p)
	}
	return out, rows.Err()
}
```

10-23 04:32에 reverse·parcel·geocode 17건이 20초~1분 25초 대기 후 같은 초에 500으로 끝났다. DB 로그에 건물 KNN 쿼리
취소가 남아 있었고, 요청 좌표는 위도 38.07~38.25 접경지였다. `ST_DWithin(geom::geography, ...)`은 GiST 인덱스 조건이
아니라서, 반경 안에 건물이 없으면 인덱스가 전국 건물을 거리순으로 끝까지 꺼낸다. 이 쿼리 몇 개가 연결 풀을 다 잡았다.

수정: 같은 쿼리에 `geom && ST_Expand(점, 반경/85km)` 상자 조건을 더했다(합성 150만 건에서 빈 지역 7.6초 -> 6ms).

```go
// RadiusDeg는 반경 m를 그 반경을 덮는 경위도 상자의 반폭(도)으로 바꾼다. 경도 1도는
// 111.32km × cos(위도)라 한국 최북단(약 38.6도)에서 약 87km이므로 85km로 나눠 넉넉히 잡는다.
//
// KNN(ORDER BY <->)에 geography 캐스트 ST_DWithin만 걸면 인덱스가 반경 조건을 못 쓴다.
// 반경 안에 건물이 하나도 없으면(바다·산·접경지) LIMIT을 채울 때까지 전국 데이터를 거리순으로
// 끝까지 훑어 수십 초가 걸리고, 연결 풀이 막혀 다른 요청까지 멈춘다(2026-10-23 운영 장애).
// geom && 상자 조건을 함께 주면 GiST가 상자 밖을 처음부터 배제한다.
func RadiusDeg(m float64) float64 { return m / 85000.0 }

// ByPoint는 입력 좌표의 역지오코딩 결과를 만든다. 도로명주소·거리·건물좌표는
// 반경 maxM 이내 최근접 건물(KNN)에서, 법정동(bcode·명)은 그 점을 포함하는 필지
// (ST_Contains)에서 얻는다. 포함 필지가 없으면(하천·미등록 등) 건물 bcode로 폴백한다.
func ByPoint(ctx context.Context, pool *pgxpool.Pool, lon, lat, maxM float64) (*Result, error) {
	var r Result
	var bldgBcode, bldgName string
	err := pool.QueryRow(ctx, `
		SELECT b.road_address, b.bcode, COALESCE(a.name,''),
		       ST_DistanceSphere(b.geom, ST_SetSRID(ST_MakePoint($1,$2),4326)),
		       ST_X(b.geom), ST_Y(b.geom)
		FROM buildings b LEFT JOIN admin_areas a ON a.bcode = b.bcode
		WHERE b.geom && ST_Expand(ST_SetSRID(ST_MakePoint($1,$2),4326), $4)
		  AND ST_DWithin(b.geom::geography, ST_SetSRID(ST_MakePoint($1,$2),4326)::geography, $3)
		ORDER BY b.geom <-> ST_SetSRID(ST_MakePoint($1,$2),4326)
		LIMIT 1`, lon, lat, maxM, RadiusDeg(maxM)).
		Scan(&r.RoadAddress, &bldgBcode, &bldgName, &r.DistanceM, &r.Point[0], &r.Point[1])
	if err != nil {
		return nil, err
	}

	// 점을 포함하는 필지의 법정동이 경계에서 최근접 건물보다 정확하다. 포함 필지가
	// 없거나 조회 실패면 건물 bcode로 폴백한다(역지오코딩 자체는 건물이 좌우).
	var parcelBcode, parcelName string
	if err := pool.QueryRow(ctx, `
		SELECT p.bcode, COALESCE(a.name,'')
		FROM parcels p LEFT JOIN admin_areas a ON a.bcode = p.bcode
		WHERE ST_Contains(p.geom, ST_SetSRID(ST_MakePoint($1,$2),4326))
		LIMIT 1`, lon, lat).Scan(&parcelBcode, &parcelName); err == nil && parcelBcode != "" {
		r.Bcode, r.AddressName = parcelBcode, parcelName
	} else {
		r.Bcode, r.AddressName = bldgBcode, bldgName
	}
	return &r, nil
}
```

```go
	api, root, err := d.endpoint()
	if err != nil {
		return nil, 0, err
	}

	total, got := -1, 0

		}

			total, _ = strconv.Atoi(strings.TrimSpace(p.TotalCount))
		}
		got += len(p.Field)

			r, ok := toRow(d, f)

			}

		}
	}
```

```go
func Run(ctx context.Context, pool *pgxpool.Pool, c *Client, o Options) (Stats, error) {
	var st Stats
	for _, d := range o.Datasets {
		dongs, err := pendingDongs(ctx, pool, d, o)
		if err != nil {
			return st, err
		}
		fmt.Fprintf(o.Log, "ned-sync %s %d: %d dongs to do\n", d, o.Year, len(dongs))
		for i, bcode := range dongs {
			if o.MaxCalls > 0 && c.Calls >= o.MaxCalls {
				st.Stopped = fmt.Sprintf("max-calls %d reached", o.MaxCalls)
				st.Calls = c.Calls
				return st, nil
			}
			rows, total, err := c.FetchDong(ctx, d, bcode, o.Year)
			if err != nil {
				var ae *APIError
				if errors.As(err, &ae) {
					st.Stopped = ae.Error() // 키·권한·한도 문제는 다음 날 이어서
					st.Calls = c.Calls
					return st, nil
				}
				return st, fmt.Errorf("%s %s: %w", d, bcode, err)
			}
			if o.Compare {
				if err := compare(ctx, pool, d, rows, &st.Cmp); err != nil {
					return st, err
				}
			} else {
				if err := upsert(ctx, pool, d, rows); err != nil {
					return st, fmt.Errorf("%s %s upsert: %w", d, bcode, err)
				}
				if _, err := pool.Exec(ctx, `
					INSERT INTO ned_sync_progress (dataset, base_year, bcode, rows) VALUES ($1,$2,$3,$4)
					ON CONFLICT (dataset, base_year, bcode) DO UPDATE SET rows = EXCLUDED.rows, synced_at = now()`,
					string(d), o.Year, bcode, len(rows)); err != nil {
					return st, err
				}
			}
			st.Dongs++
			st.Rows += len(rows)
			if (i+1)%100 == 0 || i == len(dongs)-1 {
				fmt.Fprintf(o.Log, "  %s %d/%d dongs, rows %d, calls %d (last %s total %d)\n",
					d, i+1, len(dongs), st.Rows, c.Calls, bcode, total)
			}
		}
	}
	st.Calls = c.Calls
	return st, nil
}

// pendingDongs는 필지가 실제로 있는 법정동 중 이번 연도에 아직 안 끝낸 동을 돌려준다.
// parcels(bcode, jibun) 인덱스의 범위 조건으로 시도별로 뽑는다.
func pendingDongs(ctx context.Context, pool *pgxpool.Pool, d Dataset, o Options) ([]string, error) {
	sidos := o.Sido
	if len(sidos) == 0 {
		sidos = []string{""}
	}
	var out []string
	for _, sd := range sidos {
		lo := (sd + "0000000000")[:10]
		hi := (sd + "9999999999")[:10]
		q := `SELECT DISTINCT p.bcode FROM parcels p
		      WHERE p.bcode >= $1 AND p.bcode <= $2`
		args := []any{lo, hi}
		if !o.Compare {
			q += ` AND NOT EXISTS (SELECT 1 FROM ned_sync_progress s
			        WHERE s.dataset = $3 AND s.base_year = $4 AND s.bcode = p.bcode)`
			args = append(args, string(d), o.Year)
		}
		q += ` ORDER BY 1`
		rows, err := pool.Query(ctx, q, args...)
		if err != nil {
			return nil, err
		}
		for rows.Next() {
			var b string
			if err := rows.Scan(&b); err != nil {
				rows.Close()
				return nil, err
			}
			out = append(out, b)
			if o.MaxDongs > 0 && len(out) >= o.MaxDongs {
				break
			}
		}
		rows.Close()
		if err := rows.Err(); err != nil {
			return nil, err
		}
		if o.MaxDongs > 0 && len(out) >= o.MaxDongs {
			break
		}
	}
	return out, nil
}
```

```go
package ned

import (
	"fmt"
	"time"
)

// KST는 일정 계산 기준 시간대다. distroless 이미지엔 tzdata가 없어 고정 오프셋을 쓴다(서머타임 없음).
var KST = time.FixedZone("KST", 9*60*60)

// AutoYear는 받을 기준연도다. 개별공시지가는 매년 5월 말 공시되므로 6월부터는 올해, 그 전에는 작년.
func AutoYear(now time.Time) int {
	k := now.In(KST)
	if k.Month() < time.June {
		return k.Year() - 1
	}
	return k.Year()
}

// ParseClock는 "03:17" 같은 KST 시각을 시·분으로 바꾼다.
func ParseClock(s string) (hh, mm int, err error) {
	if _, err = fmt.Sscanf(s, "%d:%d", &hh, &mm); err != nil || hh < 0 || hh > 23 || mm < 0 || mm > 59 {
		return 0, 0, fmt.Errorf("time must be HH:MM (KST): %q", s)
	}
	return hh, mm, nil
}

// NextRun은 now 이후 처음 오는 KST hh:mm이다. 지금이 정확히 그 시각이면 다음 날로 넘긴다.
func NextRun(now time.Time, hh, mm int) time.Time {
	k := now.In(KST)
	t := time.Date(k.Year(), k.Month(), k.Day(), hh, mm, 0, 0, KST)
	if !t.After(k) {
		t = t.AddDate(0, 0, 1)
	}
	return t
}
```

## 5. 운영 .env를 evnx로 관리 (10-28)

사용자가 evnx(Rust CLI, zero-knowledge .env 동기화) 자료를 주며 도입을 요청했다. 결정 사항:

- 운영 컨테이너는 evnx에 의존하지 않는다. 서버 `.env`(600)를 그대로 쓰고 evnx는 원본·이력·서버 이전 수단이다.
  evnx가 0.x, 1인 개발이라 배포를 그 서버 가용성에 묶지 않으려는 것이다.
- 마스터 비밀번호는 서버에 저장하지 않는다(`--password-stdin` 자동화 안 함). 그 비밀번호 하나가 모든 금고·버전의 열쇠다.
- `validate --fix`, `evnx update`는 쓰지 않는다(0.9.0의 비밀번호 덮어쓰기 버그, glibc 빌드로 되돌아감).
- 사용자 질문 "AES-GCM은 복호화 가능하잖아": 맞다. 복호화 키는 마스터 비밀번호에서 사용자 기기에서만 만들어지고(Argon2id),
  서버는 암호문만 가진다는 것이 zero-knowledge의 뜻이다. 비밀번호가 약하거나 새면 그대로 열린다.
- 저장소에 에이전트의 `.env` 읽기 금지 규칙(.claude/settings.json)을 넣었다. 완전한 차단이 아니라 가드레일이다.
- 대화에 붙여 넣어진 evnx 계정 비밀번호는 쓰지도 저장하지도 않았고, 사용자가 바꿨다.

설치 함정: install.sh가 주는 glibc 빌드는 GLIBC_2.39를 요구해 이 서버에서 경고가 났다 -> musl 정적 빌드를 직접 받았다.
금고 이름에는 밑줄을 쓸 수 없다(`b2d_geo` 실패 -> `b2d-geo`).

```go
type apiResp struct {
	Data []struct {
		Head []json.RawMessage `json:"head"`
		Row  []struct {
			Grp   *string  `json:"GRP_FULLNM"`
			Cls   *string  `json:"CLS_FULLNM"`
			Month string   `json:"WRTTIME_IDTFR_ID"`
			Val   *float64 `json:"DTA_VAL"`
		} `json:"row"`
	} `json:"SttsApiTblData"`
	Result *struct {
		Code    string `json:"CODE"`
		Message string `json:"MESSAGE"`
	} `json:"RESULT"`

// isNoData는 아직 공표되지 않은 달 등 "해당하는 데이터가 없습니다" 응답인지 본다.
// 운영 서버 프로브에서 문구는 확인했고 코드(INFO-200)는 공공 OpenAPI 관례라 둘 다 받는다.
func isNoData(code, msg string) bool {
	return code == "INFO-200" || strings.Contains(msg, "해당하는 데이터가 없습니다")
}

// FetchMonth는 한 달치를 모두 받는다. 아직 공표 전이면 (nil, false, nil).
func (c *Client) FetchMonth(ctx context.Context, month string) ([]Row, bool, error) {
	var out []Row
	total, got := -1, 0
	for p := 1; p <= maxPages; p++ {
		r, err := c.getPage(ctx, month, p)
		if err != nil {
			return nil, false, err
		}
		code, msg, t := pageMeta(r)
		if isNoData(code, msg) {
			if p == 1 {
				return nil, false, nil
			}
			return out, true, nil // 마지막 페이지를 넘겼다
		}
		if code != "INFO-000" {
			// 정상 응답은 항상 INFO-000을 싣는다. 코드가 없거나 다르면(키 오류 등) 멈춘다.
			return nil, false, &APIError{Msg: fmt.Sprintf("%q %s: %s", code, month, c.redact(msg))}
		}
		if total < 0 {
			total = t
		}
		n := 0
		for _, d := range r.Data {
			for _, row := range d.Row {
				n++
				if row.Val == nil || row.Cls == nil {
					continue
				}
				reg := ""
				if row.Grp != nil {
					reg = *row.Grp
				}
				out = append(out, Row{Region: reg, Zone: strings.TrimSpace(*row.Cls), Month: row.Month, Rate: *row.Val})
			}
		}
		got += n // 값이 빈 행도 센다. 건수는 원 행 기준이다
		if n == 0 || got >= total {
			return out, true, nil
		}
	}
	return nil, false, &APIError{Msg: fmt.Sprintf("%s: more than %d pages", month, maxPages)}
}

// pageMeta는 결과 코드와 전체 건수를 꺼낸다. 정상 응답은 head 안에, 오류 응답은 최상위 RESULT에 코드가 온다.
func pageMeta(r *apiResp) (code, msg string, total int) {
	if r.Result != nil {
		return r.Result.Code, r.Result.Message, 0
	}
	for _, d := range r.Data {
		for _, h := range d.Head {
			var m map[string]json.RawMessage
			if json.Unmarshal(h, &m) != nil {
				continue
			}
			if v, ok := m["list_total_count"]; ok {
				_ = json.Unmarshal(v, &total)
			}
			if v, ok := m["RESULT"]; ok {
				var res struct{ CODE, MESSAGE string }
				if json.Unmarshal(v, &res) == nil {
					code, msg = res.CODE, res.MESSAGE
				}
			}
		}
	}
	return code, msg, total
}
```

```go
package rone

import (
	"strings"

	"github.com/JO-HEEJIN/b2d_geo/internal/etl/priceindex"
)

// place는 GRP_FULLNM 하나를 해석한 결과다. Code가 있으면 그대로 쓰고, 없으면 Name을 크로스워크에서 찾는다.
type place struct {
	Name      string // "부산광역시 중구", "경기도 수원시 장안구"
	Code      string // 세종처럼 이름으로 못 찾는 곳만 직접 준다
	Legacy    bool   // "(구)동구"처럼 폐지·개편 전 이름. 같은 코드에 현재 이름 행이 있으면 그쪽이 이긴다
	Aggregate bool   // 전국·시도·시지역·군지역 같은 집계 행(적재 대상 아님)
}

// parseRegion은 API 지역 경로를 해석한다. 시도 약칭을 모르면 Name이 비고 Aggregate도 아니다(미매핑으로 센다).
//
// 확인한 모양(2026-10-28 운영 서버 응답): "부산>중구", "경기>화성시>만세구", "세종", "인천>(구)동구",
// "전남광주>광주>광산구"(광주광역시 구), "전남광주>순천시"(전라남도 시군), 집계 "서울", "경기>시지역".
func parseRegion(grp string) place {
	grp = strings.TrimSpace(grp)
	if grp == "" {
		return place{Aggregate: true} // 전국
	}
	parts := strings.Split(grp, ">")
	var p place
	for i := range parts {
		s := strings.TrimSpace(parts[i])
		if strings.HasPrefix(s, "(구)") {
			p.Legacy = true
			s = strings.TrimSpace(strings.TrimPrefix(s, "(구)"))
		}
		if s == "시지역" || s == "군지역" {
			return place{Aggregate: true}
		}
		parts[i] = s
	}
	if len(parts) == 1 {
		if parts[0] == "세종" {
			p.Code = priceindex.SejongCode // 시군구 단계가 없는 시도
			return p
		}
		return place{Aggregate: true} // 시도·권역 집계
	}

	var sido string
	rest := parts[1:]
	if parts[0] == "전남광주" {
		// R-ONE은 광주와 전남을 한 묶음으로 낸다. 주소 DB는 광주광역시·전라남도로 나뉘어 있다.
		if parts[1] == "광주" {
			if len(parts) == 2 {
				return place{Aggregate: true} // 광주 전체
			}
			sido, rest = "광주광역시", parts[2:]
		} else {
			sido = "전라남도"
		}
	} else {
		sido = priceindex.SidoFull(parts[0])
		if sido == "" {
			return place{Legacy: p.Legacy} // 모르는 시도: 미매핑
		}
	}
	p.Name = sido + " " + strings.Join(rest, " ")
	return p
}
```

레코드 만들기와 적재(internal/etl/rone/sync.go). "(구)" 행은 같은 코드에 현재 이름 행이 없을 때만 쓰고, 하위 지역이 있는
경로("경기>수원시")는 시 전체 집계로 센다. 값은 문자열로 넘겨 numeric으로 넣는다(부동소수 꼬리 방지):

```go
// build는 API 행을 (시군구, 용도지역, 월) 레코드로 바꾼다. 같은 키가 겹치면 현재 이름 행이 "(구)" 행을 이기고,
// 그 밖의 겹침은 첫 행만 둔다(한 번의 INSERT ... ON CONFLICT에 같은 키가 두 번 들어가면 실패한다).
func build(rows []Row, xw map[string]string, st *Stats, unknown map[string]bool) []rec {
	// 하위 지역이 있는 경로("경기>수원시")는 시 전체 집계다. 주소 DB에서 못 찾아도 미매핑으로 세지 않는다.
	parents := map[string]bool{}
	for _, r := range rows {
		parts := strings.Split(r.Region, ">")
		for i := 1; i < len(parts); i++ {
			parents[strings.Join(parts[:i], ">")] = true
		}
	}
	var out []rec
	at := map[string]int{}
	for _, r := range rows {
		p := parseRegion(r.Region)
		if p.Aggregate || r.Zone == "" {
			st.Aggregate++
			continue
		}
		code := p.Code
		if code == "" {
			code = xw[p.Name]
		}
		if code == "" {
			if parents[r.Region] {
				st.Aggregate++
			} else {
				st.Unmapped++
				unknown[r.Region] = true
			}
			continue
		}
		x := rec{code: code, zone: r.Zone, month: r.Month, rate: r.Rate, legacy: p.Legacy}
		k := code + "|" + r.Zone + "|" + r.Month
		i, seen := at[k]
		switch {
		case !seen:
			at[k] = len(out)
			out = append(out, x)
		case out[i].legacy && !x.legacy:
			out[i] = x
			st.Superseded++
		case x.legacy && !out[i].legacy:
			st.Superseded++
		default:
			st.Dup++
		}
	}
	return out
}

func upsert(ctx context.Context, pool *pgxpool.Pool, recs []rec) error {
	if len(recs) == 0 {
		return nil
	}
	codes := make([]string, len(recs))
	zones := make([]string, len(recs))
	months := make([]string, len(recs))
	rates := make([]string, len(recs))
	for i, r := range recs {
		codes[i], zones[i], months[i] = r.code, r.zone, r.month
		rates[i] = strconv.FormatFloat(r.rate, 'f', -1, 64) // 원값 그대로(numeric으로 넣어 부동소수 꼬리를 만들지 않는다)
	}
	_, err := pool.Exec(ctx, `
		INSERT INTO land_price_index (sigungu_code, use_zone_class, month, rate_pct, source, source_version)
		SELECT c, z, m, r::numeric, $5, now()::date
		FROM unnest($1::text[], $2::text[], $3::text[], $4::text[]) AS t(c, z, m, r)
		ON CONFLICT (sigungu_code, use_zone_class, month)
		DO UPDATE SET rate_pct=EXCLUDED.rate_pct, source=EXCLUDED.source, source_version=EXCLUDED.source_version`,
		codes, zones, months, rates, Source)
	return err
}
```

```go
// SidoFull은 R-ONE 시도 약칭을 admin_areas 정식명으로 바꾼다. 없으면 "".
func SidoFull(short string) string { return sidoFull[strings.TrimSpace(short)] }

// Crosswalk는 "시도 시군구 [자치구]" 이름 -> 시군구 5자리 코드 맵이다(다른 적재기와 공유).
func Crosswalk(ctx context.Context, pool *pgxpool.Pool) (map[string]string, error) {
	return buildCrosswalk(ctx, pool)
}

// buildCrosswalk은 admin_areas(법정동, name="시도 시군구 [자치구] 읍면동")에서
// "시도 시군구"와 "시도 시 자치구" 앞토큰 조합 → 5자리 시군구코드 맵을 만든다.
// 파일은 평시군구면 "시도 시군구", 자치구면 "시도 시 자치구"로 질의하므로 둘 다 등록한다.
// 구가 있는 시("경기도 수원시")처럼 한 이름이 여러 코드에 걸리면 그 이름은 뺀다. 그 행은 시 전체 집계라
// 어느 한 구의 코드에 넣으면 그 구의 값을 덮어쓴다(이전에는 마지막에 읽힌 구 코드로 들어갔다).
func buildCrosswalk(ctx context.Context, pool *pgxpool.Pool) (map[string]string, error) {
	rows, err := pool.Query(ctx,
		`SELECT DISTINCT substr(bcode,1,5), split_part(name,' ',1), split_part(name,' ',2), split_part(name,' ',3)
		 FROM admin_areas WHERE bcode IS NOT NULL AND name <> ''`)
	if err != nil {
		return nil, err
	}
	defer rows.Close()
	m := map[string]string{}
	ambiguous := map[string]bool{}
	put := func(k, code string) {
		if old, ok := m[k]; ok && old != code {
			ambiguous[k] = true
		}
		m[k] = code
	}
	for rows.Next() {
		var code, t1, t2, t3 string
		if err := rows.Scan(&code, &t1, &t2, &t3); err != nil {
			return nil, err
		}
		if t1 == "" || t2 == "" {
			continue
		}
		put(t1+" "+t2, code)
		if t3 != "" {
			put(t1+" "+t2+" "+t3, code)
		}
	}
	for k := range ambiguous {
		delete(m, k)
	}
	return m, rows.Err()
}
```

```go
// daily는 매일 at(KST, HH:MM)에 once를 한 번씩 돌리며 기다린다(compose 데몬용). 실패해도 죽지 않고 다음 날
// 다시 시도한다. 컨테이너 중지(SIGTERM) 시 진행 중인 단위만 버리고 끝낸다. 끝낸 단위는 DB에 남아 다음에 이어진다.
func daily(ctx context.Context, name, at string, runNow bool, once func(context.Context) error) error {
	hh, mm, err := ned.ParseClock(at)
	if err != nil {
		return err
	}
	ctx, stop := signal.NotifyContext(ctx, syscall.SIGTERM, os.Interrupt)
	defer stop()
	if runNow {
		if err := once(ctx); err != nil && ctx.Err() == nil {
			fmt.Printf("%s: run failed: %v\n", name, err)
		}
	}
	for {
		next := ned.NextRun(time.Now(), hh, mm)
		fmt.Printf("%s: next run %s\n", name, next.Format("2006-01-02 15:04 MST"))
		t := time.NewTimer(time.Until(next))
		select {
		case <-ctx.Done():
			t.Stop()
			fmt.Printf("%s: stopped\n", name)
			return nil
		case <-t.C:
		}
		if err := once(ctx); err != nil && ctx.Err() == nil {
			fmt.Printf("%s: run failed: %v\n", name, err)
		}
	}
}

// runRoneSync: b2d-etl rone-sync [--from 200501] [--to YYYYMM] [--refetch 3] [--compare]
// [--daemon --at 03:17 [--run-now]] — 한국부동산원 R-ONE Open API에서 (월) 용도지역별 지가변동률을 받아
// land_price_index에 적재한다. 처음엔 --from부터 백필하고, 이후엔 아직 없는 달과 마지막 --refetch개월만 받는다.
// --compare는 쓰지 않고 범위 안의 모든 달을 DB와 대조만 한다(검증용). 키는 env RONE_API_KEY.
func runRoneSync(ctx context.Context, args []string) error {
	fs := flag.NewFlagSet("rone-sync", flag.ContinueOnError)
	from := fs.String("from", "200501", "시작 월 YYYYMM")
	to := fs.String("to", "", "끝 월 YYYYMM. 비면 실행 시점의 이번 달(KST)")
	refetch := fs.Int("refetch", 3, "이미 받은 달 중 마지막 N개월은 다시 받음")
	pause := fs.Duration("pause", 300*time.Millisecond, "호출 사이 간격")
	cmp := fs.Bool("compare", false, "적재하지 않고 DB와 대조만")
	daemon := fs.Bool("daemon", false, "매일 --at(KST)에 실행하며 계속 대기")
	at := fs.String("at", "03:17", "--daemon 실행 시각(KST, HH:MM)")
	runNow := fs.Bool("run-now", false, "--daemon 시작 직후 한 번 먼저 실행")
	if err := fs.Parse(args); err != nil {
		return err
	}
	key := os.Getenv("RONE_API_KEY")
	if key == "" {
		return fmt.Errorf("RONE_API_KEY is empty")
	}
	pool, err := pgxpool.New(ctx, dbDSN())
	if err != nil {
		return err
	}
	defer pool.Close()

	once := func(ctx context.Context) error {
		o := rone.Options{From: *from, To: *to, Refetch: *refetch, Compare: *cmp, Log: os.Stdout}
		if o.To == "" {
			o.To = time.Now().In(ned.KST).Format("200601") // 데몬은 실행할 때마다 새 달을 범위에 넣는다
		}
		c := &rone.Client{Key: key, Pause: *pause}
		st, err := rone.Run(ctx, pool, c, o)
		fmt.Printf("rone-sync: months=%d rows=%d records=%d aggregate=%d unmapped=%d superseded=%d dup=%d calls=%d no_data=%v\n",
			st.Months, st.Rows, st.Records, st.Aggregate, st.Unmapped, st.Superseded, st.Dup, st.Calls, st.NoData)
		if len(st.Unknown) > 0 {
			fmt.Printf("rone-sync: unmapped regions: %s\n", strings.Join(st.Unknown, ", "))
		}
		if o.Compare {
			cs := st.Cmp
			fmt.Printf("compare: api_records=%d in_db=%d same=%d diff=%d only_db=%d\n", cs.APIRecords, cs.InDB, cs.Same, cs.Diff, cs.OnlyDB)
			for _, e := range cs.Examples {
				fmt.Println("  ", e)
			}
		}
		return err
	}
	if !*daemon {
		return once(ctx)
	}
	return daily(ctx, "rone-sync", *at, *runNow, once)
}
```

## 8. 함정과 실패 기록 (다시 겪지 않으려고)

1. **맥에서 서버 명령을 붙여 넣음 (2회).** ssh 줄과 서버 명령을 한 번에 붙이면 ssh가 실패한 뒤 나머지가 맥에서 돈다.
   결과는 "docker.sock 없음", "POSTGRES_PASSWORD missing". ssh 접속 후 프롬프트가 `ubuntu@ip-172-26-0-13`인지 보고 붙인다.
2. **서버 git pull 충돌.** 서버에서 base `Caddyfile`·`docker-compose.yml`을 직접 고쳐 쓰고 있었다. 백업 -> stash -> pull ->
   운영본 복원. 이후 정책: base 두 파일은 운영본 고정, 바꾸는 것은 `docker-compose.override.yml`뿐.
3. **override의 digest 고정.** 기존 override가 이미지를 digest + `pull_policy: never`로 고정해서 `latest`를 받아도 옛 앱이
   떴다. 서버 git HEAD는 실행 중인 앱 버전을 뜻하지 않는다. 버전은 override의 `sha-xxxxxxx` 태그로만 바꾼다.
4. **govulncheck CI 실패.** 화면 코드가 아니라 의존성 문제(x/text GO-2026-6629, Go 1.26.5 표준 라이브러리). Go 1.26.10,
   x/text v0.41.0, CI `check-latest`.
5. **fixture가 실데이터 문제를 가림 (2회).** 법정동 경계 NULL, 대표 용도지역 오선택. 둘 다 시험 데이터에서는 정상으로 보였다.
   운영 실주소 확인을 검증 단계에 반드시 넣는다.
6. **브이월드 파일은 서버에서 못 받음.** 200·0바이트. 로봇 로그인은 규칙 위반이라 API로 전환.
7. **python 여러 줄 명령이 목록 들여쓰기로 깨짐.** 안내 문서에서 코드 블록이 목록 안에 들어가 공백이 붙었다
   (IndentationError). 서버용 python은 한 줄로 준다.
8. **분석 한 줄에서 None 정렬 오류.** 지역 이름이 빈 행(전국 합계)이 있어 정렬이 멈췄다. `str()`로 감쌌다. 적재 코드도 같은
   행을 집계로 처리한다.
9. **evnx glibc 빌드, 금고 이름 밑줄.** musl 정적 빌드, `b2d-geo`.
10. **R-ONE 키 없이 호출.** 처음 견본 조회는 키 없는 샘플 모드(5~10건)였다. 키는 `.env`의 `RONE_API_KEY`에서 읽는다.
11. **저장소 git 신원.** 작업 환경의 전역 git 신원이 Claude로 잡혀 있었다. 커밋은 모두 JO-HEEJIN으로 했고(이력 확인),
    10-28부터 저장소 로컬 설정도 JO-HEEJIN으로 고정했다.
12. **gitnexus analyze가 CLAUDE.md·AGENTS.md를 다시 씀.** 색인 갱신의 부작용이라 되돌렸다. 색인만 갱신할 때는
    `node .gitnexus/run.cjs analyze --index-only`.
